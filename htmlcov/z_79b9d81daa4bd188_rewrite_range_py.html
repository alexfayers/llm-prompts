1"""Execute a scripted interactive rebase from a declarative JSON plan. 

2 

3Covers reorder, drop, squash, and fixup - the operations `--autosquash` 

4cannot do on its own because it only relocates commits already marked 

5`amend!`/`fixup!`/`squash!`, and cannot reorder or drop arbitrary commits. 

6 

7Plan format (a JSON array, oldest-first - the final desired top-to-bottom 

8order of the rebase todo list): 

9 

10 [ 

11 {"sha": "<full-or-abbrev-sha>", "verb": "pick"}, 

12 {"sha": "<sha>", "verb": "drop"}, 

13 {"sha": "<anchor-sha>", "verb": "pick"}, 

14 {"sha": "<sha>", "verb": "squash", "message": "final combined subject"}, 

15 {"sha": "<sha>", "verb": "fixup"} 

16 ] 

17 

18`verb` is one of: pick, drop, squash, fixup, reword. 

19`message` is optional and only meaningful on a squash/reword line - it is 

20the new subject for the commit that results from that block (the block 

21is: the preceding `pick`/`squash`/`fixup` lines up to and including this 

22one). Every squash/reword block that sets `message` gets exactly one 

23custom commit message; blocks that don't set it keep git's default 

24(concatenated) message. Message overrides are applied strictly in the 

25top-to-bottom order they appear in the plan, since that is the order the 

26rebase invokes the commit-message editor. 

27 

28Usage: 

29 python3 rewrite_range.py <plan.json> [<base-ref>] 

30 

31`base-ref` defaults to the branch's upstream (`@{u}`). Run inspect_range.py 

32first and get explicit user confirmation of the plan before invoking this. 

33""" 

34 

35import argparse 

36import json 

37import subprocess 

38import sys 

39import tempfile 

40from pathlib import Path 

41 

42VALID_VERBS = {"pick", "drop", "squash", "fixup", "reword"} 

43 

44 

45def run_git(args: list[str]) -> str: 

46 """Run a git command and return its stdout, raising on failure.""" 

47 completed = subprocess.run( 

48 ["git", *args], capture_output=True, text=True, check=True 

49 ) 

50 return completed.stdout 

51 

52 

53def resolve_base(explicit_base: str | None) -> str: 

54 """Resolve the base ref, falling back to the branch's upstream.""" 

55 if explicit_base: 

56 return explicit_base 

57 return run_git(["rev-parse", "--abbrev-ref", "@{u}"]).strip() 

58 

59 

60def load_plan(plan_path: Path) -> list[dict[str, str]]: 

61 """Load and validate the JSON plan.""" 

62 plan = json.loads(plan_path.read_text()) 

63 if not isinstance(plan, list) or not plan: 

64 raise ValueError("plan must be a non-empty JSON array") 

65 for entry in plan: 

66 if "sha" not in entry or "verb" not in entry: 

67 raise ValueError(f"plan entry missing sha/verb: {entry}") 

68 if entry["verb"] not in VALID_VERBS: 

69 raise ValueError(f"invalid verb {entry['verb']!r} in entry: {entry}") 

70 return plan 

71 

72 

73def subject_for(sha: str) -> str: 

74 """Return a commit's subject line.""" 

75 return run_git(["log", "-1", "--format=%s", sha]).strip() 

76 

77 

78def build_sequence_script(plan: list[dict[str, str]], script_path: Path) -> None: 

79 """Write a GIT_SEQUENCE_EDITOR script that emits the plan's todo list.""" 

80 lines = [ 

81 f"{entry['verb']} {entry['sha']} {subject_for(entry['sha'])}" for entry in plan 

82 ] 

83 todo = "\n".join(lines) + "\n" 

84 script_path.write_text( 

85 "#!/bin/bash\ncat > \"$1\" <<'RESCRIPT_TIDY_EOF'\n" 

86 + todo 

87 + "RESCRIPT_TIDY_EOF\n" 

88 ) 

89 script_path.chmod(0o755) 

90 

91 

92def build_message_script(plan: list[dict[str, str]], script_path: Path) -> None: 

93 """Write a GIT_EDITOR script that applies message overrides in order. 

94 

95 Each message is written to its own numbered file (rather than one 

96 newline-joined queue read back line-by-line) so multi-line messages 

97 survive intact. 

98 """ 

99 messages = [ 

100 entry["message"] 

101 for entry in plan 

102 if entry["verb"] in ("squash", "reword") and entry.get("message") 

103 ] 

104 if not messages: 

105 script_path.write_text("#!/bin/bash\ntrue\n") 

106 script_path.chmod(0o755) 

107 return 

108 

109 queue_dir = script_path.with_suffix(".queue.d") 

110 queue_dir.mkdir(exist_ok=True) 

111 for index, message in enumerate(messages, start=1): 

112 (queue_dir / f"{index}.txt").write_text(message) 

113 

114 count_file = script_path.with_suffix(".count") 

115 script = f"""#!/bin/bash 

116FILE="$1" 

117QUEUE_DIR="{queue_dir}" 

118COUNT_FILE="{count_file}" 

119COUNT=$(cat "$COUNT_FILE" 2>/dev/null || echo 0) 

120INDEX=$((COUNT + 1)) 

121MESSAGE_FILE="$QUEUE_DIR/$INDEX.txt" 

122if [ -f "$MESSAGE_FILE" ]; then 

123 cp "$MESSAGE_FILE" "$FILE" 

124 echo $((COUNT + 1)) > "$COUNT_FILE" 

125fi 

126""" 

127 script_path.write_text(script) 

128 script_path.chmod(0o755) 

129 

130 

131def build_parser() -> argparse.ArgumentParser: 

132 """Build the CLI parser.""" 

133 parser = argparse.ArgumentParser( 

134 description="Execute a scripted interactive rebase from a declarative JSON plan" 

135 ) 

136 parser.add_argument("plan", type=Path, help="Path to the JSON plan file") 

137 parser.add_argument( 

138 "base", nargs="?", default=None, help="Base ref (default: @{u})" 

139 ) 

140 return parser 

141 

142 

143def main() -> None: 

144 """Parse args, build the editor scripts, and run the rebase.""" 

145 args = build_parser().parse_args() 

146 plan = load_plan(args.plan) 

147 base = resolve_base(args.base) 

148 

149 with tempfile.TemporaryDirectory() as tmpdir: 

150 seq_script = Path(tmpdir) / "sequence_editor.sh" 

151 msg_script = Path(tmpdir) / "message_editor.sh" 

152 build_sequence_script(plan, seq_script) 

153 build_message_script(plan, msg_script) 

154 

155 env = { 

156 "GIT_SEQUENCE_EDITOR": str(seq_script), 

157 "GIT_EDITOR": str(msg_script), 

158 } 

159 result = subprocess.run( 

160 ["git", "rebase", "-i", base], 

161 env={**__import__("os").environ, **env}, 

162 check=False, 

163 ) 

164 sys.exit(result.returncode) 

165 

166 

167if __name__ == "__main__": 

168 main()
