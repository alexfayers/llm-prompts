53def _set_edits(command: Any) -> list[tuple[int, int, str]]: 

54 """Return the (start, end, text) edits that take `-e` out of a parsed `set` command, else none.""" 

55 words = [part for part in command.parts if part.kind == "word"] 

56 if not words or words[0].word != "set": 

57 return [] 

58 edits: list[tuple[int, int, str]] = [] 

59 kept = 0 

60 sets_something = False 

61 arguments = iter(words[1:]) 

62 for word in arguments: 

63 if not _SET_OPTIONS.fullmatch(word.word): 

64 rest = [word, *arguments] 

65 kept += len(rest) 

66 sets_something |= any(_LITERAL_WORD.fullmatch(w.word) for w in rest) 

67 break 

68 letters = word.word[1:] 

69 stripped = letters.replace("e", "") if word.word[0] == "-" else letters 

70 if stripped != letters: 

71 edits.append((*word.pos, f"-{stripped}" if stripped else "")) 

72 kept += bool(stripped) 

73 sets_something |= bool(stripped) or stripped == letters 

74 if "o" in letters: 

75 value = next(arguments, None) 

76 kept += value is not None 

77 if not edits: 

78 return [] 

79 if kept and not sets_something: 

80 raise _MaybeBareSet 

81 if not kept: 

82 edits.append((*words[0].pos, "true")) 

83 return edits

86def _all_set_edits(node: Any) -> list[tuple[int, int, str]]: 

87 """Return the `-e` removals of every `set` command under the parsed node.""" 

88 edits = _set_edits(node) if node.kind == "command" else [] 

89 children = [ 

90 *(getattr(node, "parts", None) or []), 

91 *(getattr(node, "list", None) or []), 

92 *([node.command] if getattr(node, "command", None) else []), 

93 *([node.output] if getattr(node, "output", None) else []), 

94 ] 

95 return edits + [edit for child in children for edit in _all_set_edits(child)]

98def _without_errexit(command: str) -> str | None: 

99 """Return the shell line without `-e` in its `set` commands, or None where it has none. 

100 

101 Raises: 

102 _MaybeBareSet: A `set -e` would be left with only words that may expand to nothing. 

103 """ 

104 import bashlex # type: ignore[import-untyped] 

105 import bashlex.errors # type: ignore[import-untyped] 

106 

107 try: 

108 edits = [ 

109 edit for node in bashlex.parse(command) for edit in _all_set_edits(node) 

110 ] 

111 except (bashlex.errors.ParsingError, NotImplementedError): 

112 return None 

113 for start, end, text in sorted(edits, reverse=True): 

114 if not text: 

115 start = len(command[:start].rstrip(" \t")) 

116 command = command[:start] + text + command[end:] 

117 return command if edits else None

144def _inline_script(name: str, words: list[str]) -> str: 

145 """Return the inline script a Python, Node or `uv run` command line runs, else an empty string.""" 

146 if name == "uv": 

147 rest = words[words.index("run") + 1 :] if "run" in words else [] 

148 program = next((i for i, word in enumerate(rest) if _runtime(word)), None) 

149 return ( 

150 "" 

151 if program is None 

152 else _inline_script(rest[program], rest[program + 1 :]) 

153 ) 

154 runtime = _runtime(name) 

155 if not runtime: 

156 return "" 

157 script_flags, value_flags = _SCRIPT_FLAGS[runtime] 

158 takes_value = False 

159 for word, after in zip(words, [*words[1:], ""]): 

160 if takes_value: 

161 takes_value = False 

162 elif word in script_flags: 

163 return after 

164 elif word[:1] != "-": 

165 return "" 

166 else: 

167 takes_value = word in value_flags 

168 return ""

171def _parses_json_inline(command: str) -> bool: 

172 """Report a shell line that runs an inline Python or Node script that parses JSON.""" 

173 import bashlex 

174 import bashlex.errors 

175 

176 try: 

177 commands = extract_commands(bashlex.parse(command)) 

178 except (bashlex.errors.ParsingError, NotImplementedError): 

179 return False 

180 return any(_JSON_CALL.search(_inline_script(c.name, c.words)) for c in commands)

184 """Report `docker compose down` with `-v` or `--volumes`, and `docker volume rm|prune`.""" 

185 compose_down = (

193class ShellRules(HooksPlugin): 

194 """Contributes command rules for `set -o errexit`, kill by port and Docker volume removal, blocks inline JSON scripts, and drops `-e` from `set`.""" 

195 

196 def on_hook( 

197 self, hook_name: str, *, logger: logging.Logger, **kwargs: object 

198 ) -> HookResult | None: 

199 """Block inline JSON scripts, else return the shell line without `-e` in its `set` calls, where the guards are active.""" 

200 if hook_name != PluginScope.PRE_SHELL or not guards_active(): 

201 return None 

202 command = cast("str", kwargs.get("command") or "") 

203 if _parses_json_inline(command): 

204 return HookResult(block=_INLINE_JSON_MESSAGE) 

205 try: 

206 fixed = _without_errexit(command) 

207 except _MaybeBareSet: 

208 return HookResult(block=_BARE_SET_MESSAGE) 

209 if fixed is None: 

210 return None 

211 return HookResult(updated_input={"command": fixed}, notes=[_ERREXIT_NOTE])

216 if not guards_active(): 

217 return []
