1"""Drop low-severity pr-review findings.""" 

2 

3import argparse 

4import json 

5import subprocess 

6import sys 

7from typing import Any 

8 

9KEEP_MAX = 3 

10SCALE = "Severity 1-5: 1 breaks behaviour or security, 2 bug or rule violation, 3 should fix, 4 minor, 5 nitpick."

13def filter_findings(findings: list[dict[str, Any]]) -> list[dict[str, Any]]: 

14 kept = [f for f in findings if f["severity"] <= KEEP_MAX] 

15 return sorted(kept, key=lambda f: f["severity"])

18def parse_findings(text: str) -> list[dict[str, Any]]: 

19 findings = json.loads(text) 

20 if not isinstance(findings, list): 

21 raise TypeError("input must be a JSON array") 

22 for finding in findings: 

23 severity = finding.get("severity") if isinstance(finding, dict) else None 

24 if type(severity) is not int or not 1 <= severity <= 5: 

25 raise ValueError(f"bad finding {finding!r}") 

26 return findings

29def run(*command: str, check: bool = True) -> subprocess.CompletedProcess[str]: 

30 return subprocess.run(list(command), capture_output=True, text=True, check=check) 

31

33def git(*args: str) -> str: 

34 return run("git", *args).stdout

37def default_branch() -> str: 

38 return git("symbolic-ref", "--short", "refs/remotes/origin/HEAD").strip()

41def pr_base() -> str: 

42 command = ["gh", "pr", "view", "--json", "baseRefName", "--jq", ".baseRefName"] 

43 try: 

44 pr = run(*command, check=False) 

45 except FileNotFoundError: 

46 pr = None 

47 if pr and pr.returncode == 0: 

48 return f"origin/{pr.stdout.strip()}" 

49 return default_branch()

52def scope() -> str: 

53 base = pr_base() 

54 sections = {"Uncommitted changes": git("diff", "HEAD")} 

55 if run("git", "rev-parse", "--abbrev-ref", "@{u}", check=False).returncode == 0: 

56 sections["Unpushed commits"] = git("diff", "@{u}...HEAD") 

57 for branch in dict.fromkeys([base, default_branch()]): 

58 sections[f"Diff against {branch}"] = git("diff", f"{branch}...HEAD") 

59 sections["Untracked files"] = git( 

60 "ls-files", "--others", "--exclude-standard", "--", ":/" 

61 ) 

62 return "".join(f"## {title}\n{body}\n" for title, body in sections.items())

65def main() -> int: 

66 parser = argparse.ArgumentParser(description=SCALE) 

67 parser.add_argument(
