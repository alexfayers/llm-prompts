44def _frontmatter_blocks(lines: Sequence[str]) -> list[_Block]: 

45 """Group frontmatter lines into keyed blocks; template lines get no key.""" 

46 blocks: list[_Block] = [] 

47 for line in lines: 

48 key = _KEY_LINE.match(line) 

49 if key: 

50 blocks.append((key.group(1), [line])) 

51 elif blocks and blocks[-1][0] and (not line.strip() or line[0] in " \t"): 

52 blocks[-1][1].append(line) 

53 else: 

54 blocks.append((None, [line]))

73def merge_agent_text(base: str, overlay: str) -> str: 

74 """Merge an overlay agent definition over a base one. 

75 

76 The overlay's frontmatter keys replace the base's and add new ones, except 

77 ``tools``, whose tokens are appended without duplicates. A non-blank 

78 overlay body replaces the base body. 

79 """ 

80 base_lines, base_body = split_frontmatter(base) or ([], base) 

81 overlay_lines, overlay_body = split_frontmatter(overlay) or ([], overlay) 

82 blocks = _frontmatter_blocks(base_lines)

84 indexes = [i for i, (k, _) in enumerate(blocks) if key and k == key] 

85 if not indexes: 

86 blocks.append((key, lines)) 

87 continue 

88 if key == "tools": 

89 lines = [_merge_tools(blocks[indexes[0]][1][0], lines[0])] 

90 blocks[indexes[0]] = (key, lines) 

91 for index in reversed(indexes[1:]): 

92 del blocks[index] 

93 frontmatter = "".join(f"{line}\n" for _, lines in blocks for line in lines) 

94 body = overlay_body if overlay_body.strip() else base_body
